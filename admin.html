var ss = SpreadsheetApp.getActiveSpreadsheet(); function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action; if (action === "newOrder") { var sheet = ss.getSheetByName("Orders") || ss.insertSheet("Orders"); if (sheet.getLastRow() === 0) { sheet.appendRow(["رقم الطلب", "التاريخ", "اسم العميل", "رقم الهاتف", "العنوان", "المنتجات", "الإجمالي", "البائع", "حالة الطلب"]); } var orderId = "ORD-" + Math.floor(100000 + Math.random() * 900000); var timestamp = new Date().toLocaleString(); // الإضافة بناءً على ترتيب الأعمدة الفعلي في صورتك sheet.appendRow([ orderId, timestamp, data.customerName, "'" + data.phone, // للحفاظ على الصفر يسار الرقم data.address, data.items, Number(data.total), data.seller || "أميل إكسبريس", "جديد" ]); updateCustomerData(data.phone, data.customerName, data.address); return ContentService.createTextOutput(JSON.stringify({status: "success", orderId: orderId})) .setMimeType(ContentService.MimeType.JSON); } else if (action === "addNewProduct") { var sheet = ss.getSheetByName("Products") || ss.insertSheet("Products"); if (sheet.getLastRow() === 0) { sheet.appendRow(["البائع", "التصنيف", "اسم المنتج", "السعر", "التصنيف الفرعي", "رابط الصورة"]); } sheet.appendRow([ data.seller || "أميل إكسبريس", data.category, data.name, Number(data.price), data.category, data.image ]); return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()})).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { var action = e.parameter ? e.parameter.action : ""; if (action === "getProducts") { var sheet = ss.getSheetByName("Products"); var products = []; if (sheet && sheet.getLastRow() > 1) { var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { products.push({ id: "P" + i, name: rows[i][2] || rows[i][0], price: rows[i][3] || rows[i][1], category: rows[i][1] || "عام", image: rows[i][5] || "", seller: rows[i][0] || "أميل إكسبريس" }); } } return ContentService.createTextOutput(JSON.stringify({status: "success", products: products})).setMimeType(ContentService.MimeType.JSON); } else if (action === "checkCustomer") { var phoneQuery = String(e.parameter.phone).replace(/\D/g, '').slice(-10); var sheet = ss.getSheetByName("Clean_Customers"); var customerData = {status: "not_found"}; if (sheet && sheet.getLastRow() > 1) { var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { var phoneVal = String(rows[i][1]).replace(/\D/g, '').slice(-10); if (phoneVal === phoneQuery && phoneQuery !== "") { customerData = { status: "found", name: rows[i][0], address: rows[i][2] }; break; } } } return ContentService.createTextOutput(JSON.stringify(customerData)).setMimeType(ContentService.MimeType.JSON); } else if (action === "getDashboardData") { var ordersSheet = ss.getSheetByName("Orders"); var productsSheet = ss.getSheetByName("Products"); var orders = []; var productsCount = productsSheet ? Math.max(0, productsSheet.getLastRow() - 1) : 0; if (ordersSheet && ordersSheet.getLastRow() > 1) { var rows = ordersSheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { // قراءة الأعمدة مطابقة تماماً لصورتك (A إلى I) orders.push({ orderId: rows[i][0] || "", date: rows[i][1] || "", customerName: rows[i][2] || "", phone: String(rows[i][3] || "").replace("'", ""), address: rows[i][4] || "", items: rows[i][5] || "", total: Number(rows[i][6]) || 0, seller: rows[i][7] || "أميل", status: rows[i][8] || "جديد" }); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", orders: orders.reverse(), productsCount: productsCount })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({status: "active"})).setMimeType(ContentService.MimeType.JSON); } function updateCustomerData(phone, name, address) { var sheet = ss.getSheetByName("Clean_Customers") || ss.insertSheet("Clean_Customers"); if (sheet.getLastRow() === 0) sheet.appendRow(["اسم العميل", "الهاتف", "العنوان", "تاريخ آخر طلب"]); var rows = sheet.getDataRange().getValues(); var cleanPhone = String(phone).replace(/\D/g, '').slice(-10); for (var i = 1; i < rows.length; i++) { if (String(rows[i][1]).replace(/\D/g, '').slice(-10) === cleanPhone) { sheet.getRange(i + 1, 1).setValue(name); sheet.getRange(i + 1, 3).setValue(address); sheet.getRange(i + 1, 4).setValue(new Date().toLocaleString()); return; } } sheet.appendRow([name, "'" + phone, address, new Date().toLocaleString()]); }
